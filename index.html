Google Sheets & Drive Integration

Auto-log rows to Google Sheets & save receipt images to Google Drive

Saved locally in your browser. Leave blank to use offline/local mode.

How to set up in 60 seconds (Free, No API Keys):

  1. Open a new or existing Google Sheet in your Google Drive.
  2. In the top menu, click Extensions → Apps Script.
  3. Delete any existing code, paste the script below, and click Save (💾).
  4. Click Deploy → New deployment → Select type Web app → Set Execute as: Me and Who has access: Anyone → Click Deploy, authorize permissions, and paste the resulting Web App URL above!
Google Apps Script (Code.gs) 
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    // Health check / Test Connection
    if (data.action === 'test') {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'success',
        message: 'Connected to Google Sheets & Drive!'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // 1. Save Receipt Image to Google Drive
    var folderName = 'SnapLedger Receipts';
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
    
    var fileUrl = '';
    if (data.imageBase64) {
      var decoded = Utilities.base64Decode(data.imageBase64);
      var safeAccount = (data.account || 'General').replace(/[^a-zA-Z0-9_-]/g, '_').substring(0, 25);
      var safeMerchant = (data.merchant || 'Receipt').replace(/[^a-zA-Z0-9_-]/g, '_').substring(0, 30);
      var fileName = safeAccount + '_' + (data.date || 'NoDate') + '_' + safeMerchant + '_' + Date.now() + '.jpg';
      var blob = Utilities.newBlob(decoded, data.mimeType || 'image/jpeg', fileName);
      var file = folder.createFile(blob);
      fileUrl = file.getUrl();
    }

    // 2. Append Record to Google Sheet
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Receipts') || ss.insertSheet('Receipts');
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['Logged At', 'Receipt Date', 'Account / Company', 'Merchant', 'Amount', 'Currency', 'Category', 'Tax Deductible', 'Notes', 'Drive File Link']);
      sheet.getRange(1, 1, 1, 10).setFontWeight('bold');
    }
    
    sheet.appendRow([
      new Date(),
      data.date || '',
      data.account || 'Personal',
      data.merchant || '',
      Number(data.amount) || 0,
      data.currency || 'MYR',
      data.category || 'Other',
      data.taxDeductible ? 'Yes' : 'No',
      data.notes || '',
      fileUrl
    ]);

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      fileUrl: fileUrl
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
